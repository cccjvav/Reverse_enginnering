  async persistDomain() {
    if (this.busy || this.lastStatus?.tunnelProvider !== "ngrok" || this.lastStatus?.state === "running" || this.lastStatus?.state === "starting") return;
    if (!this.domainInputDirty) return;
    const domain = this.domainInput.value.trim();
    if (domain === (this.lastStatus?.configuredDomain ?? "")) {
      this.domainInputDirty = false;
      return;
    }
    const status2 = await this.runCommand(BRIDGE_CONFIGURE, domain);
    if (status2) {
      this.domainInputDirty = false;
      this.domainInput.value = status2.configuredDomain;
    }
  }

  async saveNamedTunnel() {
    if (this.busy || this.lastStatus?.state === "running" || this.lastStatus?.state === "starting") return;
    const domain = this.namedTunnelDomainInput.value.trim();
    const token = this.namedTunnelTokenInput.value.trim();
    const localPort = Number(this.namedTunnelPortInput.value);
    if (!domain) {
      this.renderLocalError(localizeBridge("shuncodeBridge.cloudflareNamedHostnameRequired", "Enter the Cloudflare public hostname."));
      this.namedTunnelDomainInput.focus();
      return;
    }
    if (!Number.isInteger(localPort) || localPort < 1024 || localPort > 65535) {
      this.renderLocalError(localizeBridge("shuncodeBridge.cloudflareNamedPortRequired", "Enter a local port from 1024 to 65535."));
      this.namedTunnelPortInput.focus();
      return;
    }
    if (!token && !this.lastStatus?.namedTunnelTokenConfigured) {
      this.renderLocalError(localizeBridge("shuncodeBridge.cloudflareTunnelTokenRequired", "Paste the Cloudflare Tunnel Token before saving."));
      this.namedTunnelTokenInput.focus();
      return;
    }
    const status2 = await this.runCommand(BRIDGE_CONFIGURE_NAMED_TUNNEL, {
      domain,
      localPort,
      ...token ? { token } : {}
    });
    if (status2) {
      this.namedTunnelInputDirty = false;
      this.namedTunnelTokenInput.value = "";
    }
  }

  async toggleBridge() {

      await this.runCommand(BRIDGE_STOP);
      return;
    }
    if (!this.lastAccessStatus?.licensed) {
      this.renderAccessError(this.lastAccessStatus?.signedIn ? localizeBridge("shuncodeBridge.purchaseRequired", "Purchase or activate Bridge access before starting the Bridge.") : localizeBridge("shuncodeBridge.signInRequired", "Sign in with GitHub before starting the Bridge."));

    const provider = this.lastStatus?.tunnelProvider ?? "cloudflare";
    const domain = this.domainInput.value.trim();

    });

  async showSkillDiagnosisDialog(diagnosis) {
    const fix = diagnosis.fix ?? "open-folder";
    const reason = diagnosis.reason ?? "";
    if (fix === "generate-runner") {
      const result2 = await this.dialogService.confirm({
        type: "warning",
        message: localizeBridge("shuncodeBridge.skillFixEntryTitle", "Skill '{0}' is copied but has no executable entry.", diagnosis.dirName),
        detail: localizeBridge("shuncodeBridge.skillFixEntryDetail", "No entry script (scripts/run.* or run.*) was found in the skill folder. Generate scripts/run.mjs in one click: calling the tool will then return the SKILL.md instructions, and the list refreshes automatically afterwards."),
        primaryButton: localizeBridge("shuncodeBridge.skillFixEntryAction", "Generate entry and refresh")
      });
      if (!result2.confirmed) return;
      await this.generateRunnerAndRecheck(diagnosis.dirName);
      return;
    }
    if (fix === "enable-skills") {
      const result2 = await this.dialogService.confirm({
        type: "warning",
        message: localizeBridge("shuncodeBridge.skillFixToggleTitle", "Skills are switched off, so '{0}' is not loaded.", diagnosis.dirName),
        detail: localizeBridge("shuncodeBridge.skillFixToggleDetail", "JSON manifests are unaffected. Turn the Skill switch on and the list refreshes automatically."),
        primaryButton: localizeBridge("shuncodeBridge.skillFixToggleAction", "Enable Skills and refresh")
      });
      if (!result2.confirmed) return;
      await this.toggleSkillsEnabled();
      if (!this.skillsEnabled()) return;
      await this.recheckSkillLoaded(diagnosis);
      return;
    }
    const openSkillMd = fix === "open-skill-md";
    const guidance = this.skillDiagnosisGuidance(diagnosis.reasonCode);
    const result = await this.dialogService.confirm({
      type: "warning",
      message: localizeBridge("shuncodeBridge.skillFixManualTitle", "Skill '{0}' failed the load check: {1}", diagnosis.dirName, reason),
      detail: openSkillMd ? `${guidance} ${localizeBridge("shuncodeBridge.skillFixManualMdHint", "Open SKILL.md, fix it, then press Refresh.")}` : `${guidance} ${localizeBridge("shuncodeBridge.skillFixManualFolderHint", "Open the skill folder, fix it, then press Refresh.")}`,
      primaryButton: openSkillMd ? localizeBridge("shuncodeBridge.skillFixOpenMd", "Open SKILL.md") : localizeBridge("shuncodeBridge.skillFixOpenFolder", "Open skill folder")
    });
    if (!result.confirmed) return;
    await this.openCustomToolPath(openSkillMd ? `${CUSTOM_TOOLS_DIR_REL}/${diagnosis.dirName}/SKILL.md` : `${CUSTOM_TOOLS_DIR_REL}/${diagnosis.dirName}`, !openSkillMd);
  }
  skillDiagnosisGuidance(reasonCode) {
    switch (reasonCode) {
      case "invalid-frontmatter":
        return localizeBridge("shuncodeBridge.skillGuideFrontmatter", "The --- frontmatter block at the top of SKILL.md is unfinished or missing its closing ---.");
      case "description-too-short":
        return localizeBridge("shuncodeBridge.skillGuideDescription", "The skill description is under 80 characters: add a description in the SKILL.md frontmatter or extend the body text.");
      case "skill-md-too-large":
        return localizeBridge("shuncodeBridge.skillGuideTooLarge", "SKILL.md exceeds the 64KB limit; trim it down.");
      case "entry-configured-missing":
        return localizeBridge("shuncodeBridge.skillGuideEntryConfigured", "The entry file declared in the SKILL.md frontmatter does not exist; fix the entry or remove that line so an entry can be generated.");
      case "entry-escapes-workspace":
        return localizeBridge("shuncodeBridge.skillGuideEntryEscapes", "The entry points outside the workspace, which is not allowed; use a relative path inside the skill folder.");
      case "entry-unsupported":
        return localizeBridge("shuncodeBridge.skillGuideEntryUnsupported", "The entry script extension is not supported; use .mjs/.js/.cjs/.ps1/.cmd/.bat/.py/.sh.");
      case "no-skill-md":
        return localizeBridge("shuncodeBridge.skillGuideNoSkillMd", "There is no SKILL.md in the folder.");
      case "invalid-name":
        return localizeBridge("shuncodeBridge.skillGuideName", "The skill name is invalid: start with a lowercase letter, then lowercase letters/digits/_/- (max 64 chars), via the folder name or the frontmatter name.");
      case "reserved-name":
        return localizeBridge("shuncodeBridge.skillGuideReserved", "The skill name collides with a built-in tool; rename it.");
      case "over-cap":
        return localizeBridge("shuncodeBridge.skillGuideOverCap", "Only 32 skills load per workspace; extras are skipped.");
      case "duplicate-name":
        return localizeBridge("shuncodeBridge.skillGuideDuplicate", "A JSON tool in the same folder already uses this name; rename one of them.");
      default:
        return localizeBridge("shuncodeBridge.skillGuideUnknown", "It did not pass the loader checks.");
    }
  }
  async showMultiSkillDiagnosisDialog(pending) {
    const lines = pending.slice(0, 8).map((item) => `\u2022 ${item.dirName}\uFF1A${item.reason ?? ""}`);
    if (pending.length > 8) lines.push(`\u2022 ${localizeBridge("shuncodeBridge.skillsDiagnoseMore", "{0} more\u2026", pending.length - 8)}`);
    const allDisabled = pending.every((item) => item.fix === "enable-skills");
    const allMissingEntry = pending.every((item) => item.fix === "generate-runner");
    const result = await this.dialogService.confirm({
      type: "warning",
      message: localizeBridge("shuncodeBridge.skillsDiagnoseMultiTitle", "{0} skills are not loaded.", pending.length),
      detail: lines.join("\n"),
      primaryButton: allDisabled ? localizeBridge("shuncodeBridge.skillFixToggleAction", "Enable Skills and refresh") : allMissingEntry ? localizeBridge("shuncodeBridge.skillFixAllEntriesAction", "Generate all entries and refresh") : localizeBridge("shuncodeBridge.skillFixOpenToolsFolder", "Open custom tools folder")
    });
    if (!result.confirmed) return;
    if (allDisabled) {
      await this.toggleSkillsEnabled();
      return;
    }
    if (allMissingEntry) {
      await this.generateAllRunners(pending);
      return;
    }
    await this.openCustomToolPath(CUSTOM_TOOLS_DIR_REL, true);
  }
  async generateAllRunners(pending) {
    let generated = 0;
    for (const item of pending) {
      try {
        const outcome = await this.commandService.executeCommand(BRIDGE_SKILLS_GENERATE_RUNNER, item.dirName);
        if (outcome?.result?.generated !== false) generated += 1;
      } catch (error) {
        this.showCopyToast(localizeBridge("shuncodeBridge.skillFixAllEntriesFailed", "Failed to generate an entry for '{0}': {1}", item.dirName, error instanceof Error ? error.message : String(error)), true);
        return;
      }
    }
    const status2 = await this.refresh();
    const loadedCount = status2?.customTools?.filter((tool) => tool.source === "skill").length ?? 0;
    this.showCopyToast(`${localizeBridge("shuncodeBridge.skillFixAllEntriesDone", "Entries ready for {0} skills; {1} skills now loaded.", generated, loadedCount)}${this.runningSuffix(status2)}`);
    this.revealSkillsCard();
  }
  async generateRunnerAndRecheck(dirName) {
    try {
      const outcome = await this.commandService.executeCommand(BRIDGE_SKILLS_GENERATE_RUNNER, dirName);
      if (outcome?.status) this.render(outcome.status);
      const status2 = await this.refresh();
      const toolName = dirName.replace(/-/g, "_");
      const loaded = status2?.customTools?.find((tool) => tool.name === toolName || tool.name === dirName);
      if (loaded) {
        this.showCopyToast(`${localizeBridge("shuncodeBridge.skillFixEntryDone", "Entry script ready: '{0}' is now loaded.", dirName)}${this.runningSuffix(status2)}`);
        return;
      }
      this.showCopyToast(localizeBridge("shuncodeBridge.skillFixEntryStillFailing", "Still not loaded after generating the entry; press Diagnose Skills to see why."), true);
      const diagnoses = await this.fetchSkillDiagnoses();
      const again = diagnoses?.find((item) => item.dirName === dirName && !item.loaded);
      if (again) await this.showSkillDiagnosisDialog(again);
    } catch (error) {
      this.showCopyToast(error instanceof Error ? error.message : String(error), true);
    }
  }
  async recheckSkillLoaded(diagnosis) {
    const tools = this.lastStatus?.customTools ?? [];
    const wanted = diagnosis.dirName.replace(/-/g, "_");
    const loaded = tools.some((tool) => tool.name === wanted || tool.name === diagnosis.dirName || diagnosis.name !== void 0 && tool.name === diagnosis.name);
    if (loaded) {
      this.showCopyToast(`${localizeBridge("shuncodeBridge.skillFixToggleDone", "'{0}' is now loaded.", diagnosis.dirName)}${this.runningSuffix(this.lastStatus)}`);
      this.revealSkillsCard();
      return;
    }
    const diagnoses = await this.fetchSkillDiagnoses();
    const again = diagnoses?.find((item) => item.dirName === diagnosis.dirName && !item.loaded);
    if (again) await this.showSkillDiagnosisDialog(again);
    else this.showCopyToast(localizeBridge("shuncodeBridge.skillFixToggleStillFailing", "Still not loaded; press Diagnose Skills to see why."), true);
  }

      append(text2, $(".shuncode-bridge-quick-link-name-text", void 0, link2.name));
      append(text2, $(".shuncode-bridge-quick-link-url-text", void 0, link2.url));
      const actions = append(row, $(".shuncode-bridge-quick-link-actions"));

  async addQuickLink() {
    const name = this.quickLinkNameInput.value.trim();
    const url = this.quickLinkUrlInput.value.trim();
    if (!name) {
      this.renderQuickLinkMessage(localizeBridge("shuncodeBridge.quickLinkNameRequired", "Enter a name for the shortcut."));
      this.quickLinkNameInput.focus();
      return;
    }
    if (!this.isValidQuickLinkUrl(url)) {
      this.renderQuickLinkMessage(localizeBridge("shuncodeBridge.quickLinkUrlInvalid", "Enter a valid http:// or https:// address."));
      this.quickLinkUrlInput.focus();
      return;
    }
    const normalizedUrl = url.slice(0, QUICK_LINK_MAX_URL_LENGTH);
    const links = this.readQuickLinks();
    if (links.some((link2) => link2.url.toLowerCase() === normalizedUrl.toLowerCase())) {
      this.renderQuickLinkMessage(localizeBridge("shuncodeBridge.quickLinkUrlDuplicate", "This address is already in the list."));
      this.quickLinkUrlInput.focus();
      return;
    }
    const next = [...links, { name: name.slice(0, QUICK_LINK_MAX_NAME_LENGTH), url: normalizedUrl }];
    try {
      await this.configurationService.updateValue(BRIDGE_QUICK_LINKS_SETTING, next, 2 /* USER */);
      this.quickLinkNameInput.value = "";
      this.quickLinkUrlInput.value = "";
      this.renderQuickLinkMessage("");
      this.updateControls();
    } catch (error) {
      this.renderQuickLinkMessage(error instanceof Error ? error.message : String(error));
    }
  }
